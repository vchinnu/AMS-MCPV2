
*******Availability Status Trend ************

SapNetweaver_GetSystemInstanceList_CL
SID_s
instanceNr_d
hostname_s
dispstatus_s
Timestamp_t


let appServerList = dynamic([{ApplicationServer}]);
let baseQuery = SapNetweaver_GetSystemInstanceList_CL
| where serverTimestamp_t between (todatetime({TimeRange:start}) .. todatetime({TimeRange:end}))
| where SID_s == '{SID}'| extend instanceNr_s = tostring(toint(instanceNr_d))
| extend instanceNr_s = iff(strlen(instanceNr_s) == 2, instanceNr_s, strcat( "0", instanceNr_s))
| extend appServer = strcat(hostname_s, "_", instanceNr_s)
| where array_length(appServerList) == 0 or appServer in (appServerList)
| extend Status = case(dispstatus_s == 'SAPControl-GREEN', 0,
                        dispstatus_s == 'SAPControl-YELLOW', 1,
                        dispstatus_s == 'SAPControl-RED', 2,3 );
let timespanCount = baseQuery 
| summarize RecordCount=count()
| extend Interval = iff(RecordCount <= 10000, timespan(1m), timespan(1m))
| extend Interval = iff(RecordCount > 10000 and RecordCount < 50000, timespan(5m), Interval)
| extend Interval = iff(RecordCount >= 50000 and RecordCount < 100000, timespan(10m), Interval)
| extend Interval = iff(RecordCount >= 100000 and RecordCount < 150000, timespan(15m), Interval)
| extend Interval = iff(RecordCount >= 150000, timespan(30m), Interval)
| project Interval;
let timespanInterval = toscalar(timespanCount);
baseQuery
| summarize availability=countif(Status == 0), total=count() by bin(serverTimestamp_t, timespanInterval), appServer
| extend availability = (availability / total) * 100
| project-away total



********* Instance Process AVailability Status ************

SapNetweaver_GetProcessList_CL
description_s, 
SID_s, 
hostname_s, 
instanceNr_d, 
dispstatus_s, 
serverTimestamp_t


let appServerList = dynamic([{ApplicationServer}]);
SapNetweaver_GetProcessList_CL
| project Instance_Process = description_s, SID_s, hostname_s, instanceNr_d, dispstatus_s, serverTimestamp_t
| where serverTimestamp_t {timebrush_availability}
//commented as part of incoporating timebrush
//| where serverTimestamp_t between (todatetime({TimeRange:start}) .. todatetime({TimeRange:end}))
| where SID_s == '{SID}'| extend instanceNr_s = tostring(toint(instanceNr_d))
| extend instanceNr_s = iff(strlen(instanceNr_s) == 2, instanceNr_s, strcat( "0", instanceNr_s))
| extend appServer = strcat(hostname_s, "_", instanceNr_s)
| where array_length(appServerList) == 0 or appServer in (appServerList)
| extend Status = iff(dispstatus_s == 'SAPControl-GREEN', 0, 1)
| summarize total=count(), failureCount=countif(Status != 0) by Instance_Process, serverTimestamp_t, SID_s, appServer, instanceNr_d, hostname_s
| extend Availability = round(((toreal(total) - toreal(failureCount)) / toreal(total)) * 100, 2)
| summarize Availability = round(avg(Availability), 3), AvailabilityTrend = make_list(Availability) by Instance_Process, SID_s,hostname_s,instanceNr_d,appServer
